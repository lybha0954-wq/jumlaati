export function getDashboardPath(role: string): string {
  switch (role) {
    case "admin":
      return ROUTES.admin.home;
    case "supplier":
      return ROUTES.wholesale.overview;
    case "delivery":
      return ROUTES.delivery.overview;
    case "retailer":
    default:
      return ROUTES.retailer.overview;
  }
}
